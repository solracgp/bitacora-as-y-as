function doPost(e) { try { var data = e.parameter; // Si viene la propiedad payload como string JSON if (data.payload) { data = JSON.parse(data.payload); } var res = guardarRegistroEnHoja(data); return ContentService.createTextOutput(JSON.stringify(res)) .setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({ ok: false, error: err.toString() })) .setMimeType(ContentService.MimeType.JSON); } } function doGet(e) { return ContentService.createTextOutput("Servicio Activo AS y AS"); } function guardarRegistroEnHoja(data) { var ss = SpreadsheetApp.getActiveSpreadsheet(); var hojaVisitas = ss.getSheetByName("VISITAS") || ss.insertSheet("VISITAS"); if (hojaVisitas.getLastRow() === 0) { hojaVisitas.appendRow([ "Correlativo", "Fecha", "Cliente", "Local", "Sigla", "Técnico", "Tipo Visita", "Hora Inicio", "Hora Término", "Resumen Hallazgos", "Observaciones", "Links Fotos" ]); } var prefijo = getPrefijoTecnico(data.tecnico); var numFila = hojaVisitas.getLastRow(); // Mantiene el conteo ordenado var numeroSecuencial = ("000" + numFila).slice(-3); var correlativoSecuencial = prefijo + "-" + numeroSecuencial; var fotosUrls = []; if (data.hallazgos) { var hallazgosArr = typeof data.hallazgos === 'string' ? JSON.parse(data.hallazgos) : data.hallazgos; hallazgosArr.forEach(function(h, index) { if (h.fotoBase64) { var urlFoto = guardarFotoEnDrive(h.fotoBase64, correlativoSecuencial + "_Foto_" + (index + 1)); if (urlFoto) fotosUrls.push(urlFoto); } }); } var resumenTexto = data.resumen; if (!resumenTexto && data.hallazgos) { var hallazgosArr = typeof data.hallazgos === 'string' ? JSON.parse(data.hallazgos) : data.hallazgos; resumenTexto = hallazgosArr.map(function(h){ return h.problema; }).join(" | "); } hojaVisitas.appendRow([ correlativoSecuencial, data.fecha || new Date().toISOString().split('T')[0], data.cliente, data.local, data.sigla, data.tecnico, data.tipo, data.horaInicio, data.horaTermino, resumenTexto, data.observ, fotosUrls.join("\n") ]); return { ok: true, correlativo: correlativoSecuencial }; } function getPrefijoTecnico(nombre) { if (!nombre) return "REG"; if (nombre.indexOf("GONZALEZ") !== -1) return "CG"; if (nombre.indexOf("PRUEBA") !== -1) return "CP"; if (nombre.indexOf("BENJAMIN") !== -1) return "BA"; return "REG"; } function guardarFotoEnDrive(base64Data, nombreArchivo) { try { var carpeta = DriveApp.getFoldersByName("Fotos_Visitas_AS_Y_AS"); var folder = carpeta.hasNext() ? carpeta.next() : DriveApp.createFolder("Fotos_Visitas_AS_Y_AS"); var splitBase = base64Data.split(","); var contentType = splitBase[0].match(/:(.*?);/)[1]; var bytes = Utilities.base64Decode(splitBase[1]); var blob = Utilities.newBlob(bytes, contentType, nombreArchivo + ".jpg"); var archivo = folder.createFile(blob); archivo.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); return archivo.getUrl(); } catch (e) { return null; } }
